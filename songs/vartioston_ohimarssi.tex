\begin{song}{Vartioston ohimarssi}

	\repetitionbegin{}
	La garde passe; il est minuit.
	Qu'on se retire et plus de bruit.

	La garde passe et la voici:
	Rentrez en diligence.
	obéisez, faites silence.
	C'est la loi du Cadi.
	\repetitionend{}

	\repetitionbegin{}
	Qu'on se retire et plus de bruit.
	La garde passe, il est minuit.
	Plus de bruit, plus de bruit!

	Que tout setaiseici,
	Rentrez chez vous en diligence.
	Obéisez, faites silence.
	C'est la loi du Cadi.
	\repetitionend{}

	\repetitionbegin{}
	Vartio menee ohi, on keskiyö.
	Kun vetäydymme, ei enää melua.

	Vartio menee ohi, ja katsokaa:
	Vetäytykää rauhassa.
	totelkaa, olkaa hiljaa.
	Se on Cadin laki.
	\repetitionend{}

	\repetitionbegin{}
	Kun vetäydymme, ei enää melua.
	Vartio menee ohi, on keskiyö.
	Ei enää melua, ei enää melua!

	Kaikki jotka olivat täällä,
	menkää kotiinne rauhassa.
	Totelkaa, olkaa hiljaa.
	Se on Cadin laki.
	\repetitionend{}

\end{song}
