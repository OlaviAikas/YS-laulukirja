\begin{song}{YS-kissan banjo}

    \original{\citetitle{mustan_kissan_tango}}

	La-la-la-la-la-la lal-la
	Jos YS-kissan saisin mä sitä rakastaisin
	Sen YS-kissan tassut ne sitten hassut on
	Kun pistän narun päihin mä hiekkalapion
	niin kissan oma tangobanjo valmis on!

	Hei näytä kissa banjo banjo banjo
	nyt YS-kissan soitin sen ikioma on!
	Soi YS-kissan banjo banjo banjo
	sen sulosointi raikaa tänne meillekin
	La-la-la-la-la-la lal-la

	On YS-kissa Tilli ja YS-kissa Tupli
	ja YS-kissa Tosca ja Tyyne, Manteli
	On Justiina ja Juudas ja Reha, Stögäkin,
	ja Hilde, Pepsi, Puuro, lienee muitakin? *

	Hei näytä kissa banjo banjo banjo
	nyt YS-kissan soitin sen ikioma on!
	Soi YS-kissan banjo banjo banjo
	sen sulosointi raikaa tänne meillekin
	La-la-la-la-la-la lal-la

	Sen YS-kissan päivät on aidot kissan päivät
	sen YS-kissan posket jo ovat säämiskää
	Se YS-kissa tähti on YLE-radion
	Sen banjo käsin valmistettu tuote on!

	Hei näytä kissa banjo banjo banjo
	nyt YS-kissan soitin sen ikioma on!
	Soi YS-kissan banjo banjo banjo
	sen sulosointi raikaa tänne meillekin –
	Hei näytä kissa banjo banjo banjo
	nyt YS-kissan soitin sen ikioma on
	Teostorikkauksissa on kohta YS-kissa
	ja siitä kissa kiittääää – nyt kaikkein kauneimmin
	La-la-la-la-la-la lal-la Miau!

	\vspace{2em}

	* \textit{Jos YS-kissasi puuttuu laulusta, se johtuu siitä,
	ettei Liina tunne tätä. Ilmianna YS-kissasi Liinalle niin korjataan asia!}
\end{song}
